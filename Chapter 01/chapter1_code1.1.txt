# Create a resource groupaz group create --name myResourceGroup --location eastus# Create AKS cluster with 2 nodesaz aks create --resource-group myResourceGroup --name myAKSCluster --node-count 2 --enable-addons monitoring --generate-ssh-keys# Get credentials to interact with the clusteraz aks get-credentials --resource-group myResourceGroup --name myAKSCluster# Deploy an application (e.g., nginx) to the clusterkubectl create deployment nginx --image=nginx# Expose deployment on port 80kubectl expose deployment nginx --port=80 --type=LoadBalancer# Check service statuskubectl get services
